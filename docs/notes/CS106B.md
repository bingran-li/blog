const string VOWELS = "aeiouAEIOU";

string onlyConnectize(string phrase) {
    string answer = "";
    if (phrase.length() == 1) {
        if (isalpha(phrase[0])) {
            if (VOWELS.find(phrase) == string::npos) {
                return toUpperCase(phrase);
            } else {
                return "";
            }
        } else {
                return "";
            }
    }
    answer=onlyConnectize(phrase.substr(0,phrase.length()-1))+onlyConnectize(phrase.substr(phrase.length()-1,1));
    return answer;
}

#include "Plotter.h"
#include "strlib.h"
using namespace std;

void runPlotterScript(istream& input) {
    PenStyle pen = {1, "black"};
    double x0=0, y0=0, x1, y1, dx, dy;
    string op;
    bool bl = 0;
    while (input >> op) {
        op=toLowerCase(op);
        if (op == "pendown") {
            bl=1;
        } else if (op == "penup"){
            bl=0;
        } else if (op == "moveabs"){
            input>>x1>>y1;
            if (bl) {
                drawLine(x0,y0,x1,y1, pen);
            }
            x0 = x1;
            y0 = y1;
        } else if (op == "moverel") {
            input>>dx>>dy;
            x1=x0+dx; y1=y0+dy;
            if (bl) {
                drawLine(x0, y0, x1, y1, pen);
            }
            x0 = x1;
            y0 = y1;
        } else if (op == "pencolor") {
            input>>pen.color;
        } else if (op == "penwidth") {
            input>>pen.width;
        }
    }
}
